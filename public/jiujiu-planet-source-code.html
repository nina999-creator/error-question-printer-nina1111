30// Timeout wrapper to ensure requests NEVER hang for minutes
31async function callWithTimeout<T>(promise: Promise<T>, ms = 13000): Promise<T> {
32  let timeoutId: any;
33  const timeoutPromise = new Promise<never>((_, reject) => {
34    timeoutId = setTimeout(() => {
35      reject(new Error(`AI 服务响应超时（超过${Math.round(ms / 1000)}秒），已启动自动保护`));
36    }, ms);
37  });
38  return Promise.race([promise, timeoutPromise]).finally(() => clearTimeout(timeoutId));
39}

42async function generateContentWithFallback(params: any, preferredModels = ['gemini-3.1-flash-lite', 'gemini-3.8-flash']) {
43  let lastError: any = null;
44

81function formatErrorMessage(err: any, defaultMsg: string): string {

83  const rawMsg = err.message || String(err);

93      }
94      return parsed.error.message;

97    // not raw JSON
98  }

101    return 'AI 识题通道短时繁忙，请稍候片刻点击重试。';

1/**
2 * Client-side image compression and downscaling utility.
3 * Downscales smartphone photos from 10MB-20MB down to ~300KB (max 1600px, 0.85 quality),
4 * which speeds up upload and AI OCR processing by 10x-50x while preserving formula clarity.
5 */
6export async function compressImage(
7  source: File | string,
8  maxWidth = 1600,
9  maxHeight = 1600,
10  quality = 0.85
11): Promise<{ dataUrl: string; mimeType: string; sizeKb: number }> {
12  return new Promise((resolve, reject) => {
13    const img = new Image();
14
15    img.onload = () => {
16      let { width, height } = img;
17
18      // Calculate new dimensions while keeping aspect ratio
19      if (width > maxWidth || height > maxHeight) {
20        if (width / height > maxWidth / maxHeight) {
21          height = Math.round((height * maxWidth) / width);
22          width = maxWidth;
23        } else {
24          width = Math.round((width * maxHeight) / height);
25          height = maxHeight;
26        }
27      }
28
29      const canvas = document.createElement('canvas');
30      canvas.width = width;
31      canvas.height = height;
32
33      const ctx = canvas.getContext('2d');
34      if (!ctx) {
35        // Fallback to original
36        if (typeof source === 'string') {
37          return resolve({ dataUrl: source, mimeType: 'image/jpeg', sizeKb: Math.round(source.length / 1024) });
38        }
39        return reject(new Error('Canvas context not available'));
40      }
41
42      // Smooth resizing
43      ctx.imageSmoothingEnabled = true;
44      ctx.imageSmoothingQuality = 'high';
45      ctx.drawImage(img, 0, 0, width, height);
46
47      const mimeType = 'image/jpeg';
48      const dataUrl = canvas.toDataURL(mimeType, quality);
49      const sizeKb = Math.round((dataUrl.length * 3) / 4 / 1024);
50
51      resolve({ dataUrl, mimeType, sizeKb });
52    };
53
54    img.onerror = (err) => {
55      console.warn('Failed to load image for compression:', err);
56      if (typeof source === 'string') {
57        resolve({ dataUrl: source, mimeType: 'image/jpeg', sizeKb: 0 });
58      } else {
59        reject(new Error('无法读取图片，请检查图片格式'));
60      }
61    };
62
63    if (typeof source === 'string') {
64      img.src = source;
65    } else {
66      const reader = new FileReader();
67      reader.onload = (e) => {
68        img.src = e.target?.result as string;
69      };
70      reader.onerror = reject;
71      reader.readAsDataURL(source);
72    }
73  });
74}

6export function renderHighlightedText(text: string, keywords: string[] = []): React.ReactNode {
7  if (!text) return null;

10  }
11
12  // Filter valid non-empty keywords and escape regex characters
13  const validKeywords = keywords
14    .map((k) => k.trim())
15    .filter((k) => k.length > 0)
16    .sort((a, b) => b.length - a.length); // match longest first
17

20  }

24
25  const parts = text.split(regex);
26

31        if (isMatch) {

38            </mark>
